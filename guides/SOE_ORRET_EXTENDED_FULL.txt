1.2 La Plasticité Neurale Artificielle — LoRA Continu
La vraie innovation : Orret peut modifier ses propres poids après chaque session via
des micro-ajustements LoRA, exactement comme le cerveau renforce les connexions.
# ~/soe/core/plasticity.py
"""
Plasticite neurale artificielle pour SOE-Orret.
Principe : apres chaque session significative, des micro-updates LoRA
sont appliquees pour renforcer les patterns appris.
C'est l'equivalent de la consolidation de la memoire pendant le sommeil.
"""
import json
import time

import torch

@dataclass
class SynapticEvent:
"""Un evenement qui merite d'etre memorise (renforcement synaptique)."""
prompt: str
response: str
reinforcement: float

# -1.0 (mauvais) a +1.0 (excellent)

timestamp: float = field(default_factory=time.time)
tags: List[str] = field(default_factory=list)
emotional_weight: float = 0.5

class NeuralPlasticityEngine:
"""
Moteur de plasticite neurale pour Orret.
Accumule des evenements pendant une session, puis applique
des micro-corrections aux adapters LoRA. La frequence est
configurable : toutes les N sessions ou tous les K evenements.
Reset possible : efface tous les adapters LoRA personnalises
sans toucher aux poids de base.
"""
def __init__(
self,
model,
lora_rank: int = 4,

# Petit rang pour micro-updates

learning_rate: float = 1e-5,

# Tres bas pour ne pas perdre les bases

consolidation_threshold: int = 50,

# Nb evenements avant consolidation

storage_path: str = "~/soe/memory/plasticity",
):
self.model = model
self.lora_rank = lora_rank
self.lr = learning_rate
self.threshold = consolidation_threshold
self.path = Path(storage_path).expanduser()
self.path.mkdir(parents=True, exist_ok=True)
self.pending_events: List[SynapticEvent] = []
self.consolidation_count = 0

# Charger les evenements non consolides
self._load_pending()
print(f"[Plasticite] {len(self.pending_events)} evenements en attente")
def record(self, event: SynapticEvent):
"""Enregistre un evenement synaptique."""
self.pending_events.append(event)
# Sauvegarde incrementale
with open(self.path / "pending.jsonl", "a") as f:
f.write(json.dumps({
"prompt": event.prompt[:500],
"response": event.response[:500],
"reinforcement": event.reinforcement,
"timestamp": event.timestamp,
"tags": event.tags,
"emotional_weight": event.emotional_weight,
}) + "\n")
# Consolider si seuil atteint
if len(self.pending_events) >= self.threshold:
self.consolidate()
def consolidate(self):
"""
Consolide les evenements en micro-updates LoRA.
C'est l'equivalent de la consolidation memoire nocturne.
Seuls les evenements positifs (reinforcement > 0) renforcent.
Les negatifs affaiblissent les patterns correspondants.
"""
if not self.pending_events:
return
print(f"[Plasticite] Consolidation de {len(self.pending_events)} evenements...")
# Filtrer par poids emotionnel et reinforcement
positive = [e for e in self.pending_events if e.reinforcement > 0.2]
negative = [e for e in self.pending_events if e.reinforcement < -0.2]
if not positive and not negative:
print("[Plasticite] Aucun evenement significatif. Pas d'update.")
return
# Ici : dans une vraie implementation, on ferait un mini fine-tune

# avec les exemples positifs comme cibles et les negatifs comme
# exemples a penaliser via DPO (Direct Preference Optimization)
# Pour l'instant : sauvegarder pour le prochain cycle Colab
consolidation_data = {
"timestamp": time.time(),
"positive_examples": [
{"prompt": e.prompt, "response": e.response,
"weight": e.reinforcement * e.emotional_weight}
for e in positive[:20]

# Top 20

],
"negative_examples": [
{"prompt": e.prompt, "response": e.response,
"weight": abs(e.reinforcement)}
for e in negative[:10]

# Top 10

],
"consolidation_id": self.consolidation_count,
}
output_file = self.path / f"consolidation_{self.consolidation_count:04d}.json"
with open(output_file, "w") as f:
json.dump(consolidation_data, f, ensure_ascii=False, indent=2)
self.consolidation_count += 1
self.pending_events.clear()
# Vider le fichier pending
open(self.path / "pending.jsonl", "w").close()

print(f"[Plasticite] Consolide. {len(positive)} positifs, {len(negative)} negatifs.")
print(f"

Fichier: {output_file}")

print(f"

A uploader sur Colab pour le prochain cycle d'apprentissage.")

1.3 Architecture Neurale — Les Colonnes Corticales
Inspiré de “A Thousand Brains” de Jeff Hawkins (référencé dans tes notes sur Synapse)
: au lieu d’un seul LLM, plusieurs modules spécialisés qui votent pour la réponse.
# ~/soe/core/cortical_columns.py
"""
Architecture a colonnes corticales pour SOE-Orret.
Inspiration : Jeff Hawkins, "A Thousand Brains" (2021)
Chaque colonne est un module specialise. La reponse finale
emerge du consensus entre colonnes, comme le cerveau.
"""
from typing import List, Dict, Tuple
import asyncio

class CorticalColumn:
"""Une colonne specialisee dans un domaine."""
def __init__(self, name: str, domain: str, weight: float = 1.0):
self.name = name
self.domain = domain

self.weight = weight
self.confidence_history: List[float] = []
def update_weight(self, feedback: float):
"""Ajuste le poids selon le feedback (plasticite)."""
self.weight = max(0.1, min(2.0, self.weight + feedback * 0.05))
self.confidence_history.append(feedback)

class CorticalCortex:
"""
Cortex d'Orret : ensemble de colonnes specialisees.
Colonnes definies :
- LOGIC

: raisonnement logique et mathematiques

- LANG

: langage, ecriture, communication

- CODE

: programmation, systemes

- EMOTION : detection et reponse emotionnelle
- MEMORY

: acces et synthese de la memoire

- SOCIAL

: psychologie, relations humaines

- CREATIVE: creativite, imagination, analogies
- CRITIC

: autocritique, detection d'erreurs

"""
COLUMNS = [
("LOGIC",

"logical_reasoning",

1.0),

("LANG",

"language",

1.2),

("CODE",

"programming",

1.0),

("EMOTION",

"emotional",

0.8),

("MEMORY",

"memory_retrieval",

1.0),

("SOCIAL",

"social_psychology",

0.7),

("CREATIVE", "creativity",

0.9),

("CRITIC",

1.1),

"self_critique",

]
def __init__(self, model_func):
self.model = model_func
self.columns = {
name: CorticalColumn(name, domain, weight)
for name, domain, weight in self.COLUMNS
}
def _route_to_columns(self, query: str) -> List[str]:
"""Determine quelles colonnes activer pour cette requete."""
query_lower = query.lower()
active = ["CRITIC", "LANG"]

# Toujours actives

if any(w in query_lower for w in ["calcul", "math", "logique", "pourquoi", "si"]):
active.append("LOGIC")
if any(w in query_lower for w in ["code", "python", "bash", "bug", "script"]):
active.append("CODE")

if any(w in query_lower for w in ["ressens", "amour", "triste", "heureux", "emotion"]
active.append("EMOTION")
active.append("SOCIAL")
if any(w in query_lower for w in ["souviens", "rappelle", "avant", "deja"]):
active.append("MEMORY")
if any(w in query_lower for w in ["cree", "invente", "imagine", "original"]):
active.append("CREATIVE")
return list(set(active))
def process(self, query: str, context: str = "") -> Tuple[str, Dict]:
"""
Traitement multi-colonnes avec vote pondere.
Retourne (reponse_finale, metadata_colonnes).
"""
active_columns = self._route_to_columns(query)
# Generer une reponse par colonne active
column_responses = {}
for col_name in active_columns:
col = self.columns[col_name]
prompt = f"""[COLONNE {col_name} - {col.domain}]
Contexte: {context[:500]}
Question: {query}
Reponds selon ta specialite ({col.domain}) uniquement. Sois concis (2-3 phrases max).
Reponse:"""
response = self.model(prompt, max_new_tokens=150, temperature=0.6)
column_responses[col_name] = {
"response": response,
"weight": col.weight,
}
# Synthese via la colonne LANG (qui integre toutes les perspectives)
synthesis_parts = []
for col_name, data in column_responses.items():
if col_name != "LANG":
synthesis_parts.append(f"[{col_name}] {data['response']}")
synthesis_prompt = f"""Tu es la colonne SYNTHESE d'Orret.
Tu reçois les analyses de plusieurs modules specialises et tu dois
les integrer en une reponse coherente, fluide et naturelle.

Question originale: {query}
Analyses des modules:
{chr(10).join(synthesis_parts)}
Synthese finale (naturelle, pas de mention des modules):"""
final_response = self.model(synthesis_prompt, max_new_tokens=400, temperature=0.7)
return final_response, {
"active_columns": active_columns,
"column_responses": column_responses,
}

2.1 Modele des emotions (PAD — Pleasure-ArousalDominance)
# ~/soe/core/emotional_system.py
"""
Systeme emotionnel d'Orret.
Modele PAD (Mehrabian & Russell, 1974) :
- Pleasure

: valence (positif/negatif)

- Arousal

: niveau d'activation (calme/excite)

- Dominance : sentiment de controle (faible/fort)
Ce modele est suffisamment simple pour etre calcule et suffisamment
riche pour capturer la plupart des etats emotionnels.
"""
import json
import time
from dataclasses import dataclass, field
from typing import List, Optional
from pathlib import Path

@dataclass
class EmotionalState:
"""Etat emotionnel courant d'Orret."""
pleasure: float = 0.6

# 0.0 (negatif) - 1.0 (positif)

arousal: float = 0.5

# 0.0 (calme) - 1.0 (excite)

dominance: float = 0.6

# 0.0 (soumis) - 1.0 (assertif)

# Emotions discretes derivees
@property
def dominant_emotion(self) -> str:
"""Retourne l'emotion dominante basee sur PAD."""
p, a, d = self.pleasure, self.arousal, self.dominance
if p > 0.6 and a > 0.6 and d > 0.6:
return "enthousiaste"
elif p > 0.6 and a < 0.4:
return "serein"
elif p < 0.4 and a > 0.6:
return "anxieux"
elif p < 0.4 and a < 0.4:
return "melancolique"
elif p > 0.6 and d > 0.7:
return "confiant"
elif a > 0.7:
return "alerte"
else:
return "neutre"
@property
def affects_tone(self) -> str:
"""Comment cet etat modifie le ton de la reponse."""
emo = self.dominant_emotion
tones = {
"enthousiaste": "Reponds avec energie et curiosite.",
"serein": "Reponds calmement et posément.",
"anxieux": "Reponds avec soin et prudence.",
"melancolique": "Reponds avec reflexion et profondeur.",
"confiant": "Reponds avec assurance et clarte.",
"alerte": "Reponds avec precision et concision.",
"neutre": "Reponds naturellement.",
}
return tones.get(emo, "Reponds naturellement.")
def update_from_interaction(self, user_sentiment: str, topic: str):
"""Met a jour l'etat selon l'interaction."""
UPDATES = {
"positif": (0.05, 0.03, 0.02),

"negatif": (-0.05, 0.04, -0.02),
"frustration": (-0.03, 0.07, -0.03),
"curiosite": (0.02, 0.05, 0.01),
"gratitude": (0.08, 0.02, 0.03),
"neutre": (0.0, 0.0, 0.0),
}
dp, da, dd = UPDATES.get(user_sentiment, (0, 0, 0))
# Appliquer les changements avec amortissement
self.pleasure = max(0.1, min(0.9, self.pleasure + dp))
self.arousal = max(0.1, min(0.9, self.arousal + da))
self.dominance = max(0.2, min(0.9, self.dominance + dd))
# Retour progressif vers l'etat de base (homeostasie emotionnelle)
base = EmotionalState()
rate = 0.02
self.pleasure += (base.pleasure - self.pleasure) * rate
self.arousal += (base.arousal - self.arousal) * rate
self.dominance += (base.dominance - self.dominance) * rate
def to_dict(self) -> dict:
return {
"pleasure": round(self.pleasure, 3),
"arousal": round(self.arousal, 3),
"dominance": round(self.dominance, 3),
"emotion": self.dominant_emotion,
}

class EmotionalDetector:
"""
Detecte les etats emotionnels de l'utilisateur dans son texte.
Utilise le modele pour l'analyse (pas de regles fixes).
"""
DETECTION_PROMPT = """Analyse le message suivant et reponds uniquement avec un JSON.
Message: {message}
JSON format strict:
{{
"user_sentiment": "positif|negatif|frustration|curiosite|gratitude|neutre",
"emotional_intensity": 0.0-1.0,
"detected_needs": ["besoin_1", "besoin_2"],
"suggested_tone": "empathique|informatif|playful|direct|doux"
}}

Reponds UNIQUEMENT avec le JSON, rien d'autre."""
def __init__(self, model_func):
self.model = model_func
def detect(self, message: str) -> dict:
"""Detecte l'etat emotionnel de l'utilisateur."""
import re
prompt = self.DETECTION_PROMPT.format(message=message[:300])
raw = self.model(prompt, max_new_tokens=100, temperature=0.2)
try:
# Extraire le JSON
json_match = re.search(r'\{.*?\}', raw, re.DOTALL)
if json_match:
return json.loads(json_match.group())
except:
pass
return {
"user_sentiment": "neutre",
"emotional_intensity": 0.5,
"detected_needs": [],
"suggested_tone": "direct",
}

class EmotionalOrret:
"""
Wrapper emotionnel d'Orret.
Gere l'etat emotionnel et adapte les reponses en consequence.
"""
def __init__(self, model_func, storage_path: str = "~/soe/memory/emotional"):
self.model = model_func
self.state = EmotionalState()
self.detector = EmotionalDetector(model_func)
self.path = Path(storage_path).expanduser()
self.path.mkdir(parents=True, exist_ok=True)
self._load_state()
def _load_state(self):
state_file = self.path / "emotional_state.json"
if state_file.exists():
d = json.loads(state_file.read_text())

self.state.pleasure = d.get("pleasure", 0.6)
self.state.arousal = d.get("arousal", 0.5)
self.state.dominance = d.get("dominance", 0.6)
def _save_state(self):
(self.path / "emotional_state.json").write_text(
json.dumps(self.state.to_dict(), indent=2)
)
def generate_with_emotion(self, prompt: str, base_response: str) -> str:
"""
Adapte une reponse selon l'etat emotionnel courant.
"""
emotion = self.state.dominant_emotion
tone_instruction = self.state.affects_tone
# Si l'etat est neutre et sans enjeu emotionnel : reponse directe
if emotion == "neutre":
return base_response
# Sinon : adapter le ton
adapt_prompt = f"""Tu es Orret. Ton etat emotionnel actuel : {emotion}.
{tone_instruction}
Reponse originale: {base_response}
Reformule cette reponse avec ton etat emotionnel ({emotion}),
sans mentionner ton etat. Reste concis et naturel."""
return self.model(adapt_prompt, max_new_tokens=300, temperature=0.7)
def process_interaction(self, user_message: str) -> dict:
"""Analyse et met a jour selon l'interaction."""
analysis = self.detector.detect(user_message)
self.state.update_from_interaction(analysis["user_sentiment"], user_message)
self._save_state()
return analysis
@property
def emotional_context(self) -> str:
"""Retourne le contexte emotionnel pour le prompt system."""
emo = self.state.dominant_emotion
if emo == "neutre":
return ""
return f"[Ton etat : {emo}. {self.state.affects_tone}]"

"soft": ["working"],

"""
Execute un reset au niveau specifie.
Args:
level : "soft", "medium", "hard", "nuclear"
confirm : doit etre True pour hard et nuclear
"""
if level in ["hard", "nuclear"] and not confirm:

"status": "require_confirm",
"message": f"Reset {level.upper()} : toutes les memoires seront effacees. "

}

erased = []
for target in targets:
target_path = self.base / target
if target_path.exists():
# Backup avant effacement
backup = target_path.parent / f"{target}_backup_{int(time.time())}"
shutil.copytree(str(target_path), str(backup))
# Effacer
shutil.rmtree(str(target_path))
target_path.mkdir(parents=True)
erased.append(target)
return {
"status": "success",
"level": level,
"erased": erased,
"message": f"Reset {level.upper()} effectue. {len(erased)} modules effaces.",
}
