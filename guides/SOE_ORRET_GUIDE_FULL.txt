PARTIE 0 — Philosophie et Architecture
0.1 Pourquoi un dLLM et pas un LLM classique
Tous les LLMs actuels sont autoregressifs (AR) : ils génèrent les tokens un par un, de
gauche à droite. C’est comme écrire une phrase sans jamais pouvoir revenir en arrière.
Problèmes fondamentaux de l’AR :
Génération séquentielle impossible à paralléliser
Le Reversal Curse : incapacité à raisonner en sens inverse
Left-to-right bias : chaque token conditionné sur le passé uniquement
Pas de révision possible après génération
Un dLLM fonctionne différemment. Il commence avec une séquence masquée et
débruite progressivement en remplissant tous les tokens en parallèle :
Étape 0 : [MASK][MASK][MASK][MASK][MASK][MASK][MASK][MASK]
Étape 1 : [MASK] suis [MASK][MASK] assistant[MASK][MASK][MASK]

Avantages fondamentaux :
Contexte bidirectionnel : chaque token voit TOUS les autres (passé et futur)
Génération parallèle : tous les tokens d’un bloc générés simultanément
Révision possible : tokens peu confiants re-maskés et regénérés
Résout le Reversal Curse : LLaDA 8B (2025) surpasse GPT-4o sur les tâches
inverses

0.2 Pourquoi A2D et pas from-scratch
Entraîner un LLM from scratch coûte des millions. La conversion A2D (Autoregressiveto-Diffusion) transforme un modèle AR existant en dLLM via continued pretraining en
préservant tout le savoir déjà appris.
La recette SCP (Simple Continual Pretraining) de RND1/Fast-dLLM v2 (2025) :
1. Partir d’un checkpoint AR fort (Qwen2.5-7B)
2. Remplacer le masque causal par un masque bidirectionnel
3. Continuer l’entraînement avec l’objectif de démasquage MDLM
4. Learning rates différenciés : élevé pour attention, bas pour FFN (évite le catastrophic
forgetting)
Fast-dLLM v2 a adapté Qwen2.5-7B en ~12h sur 64 A100s. Avec Google Colab Pro
(A100), ça prend 6-15h selon la taille du dataset.

PARTIE 3 — Conversion A2D : AR vers dLLM
3.1 Theorie — MDLM et Block Diffusion
AR classique : predire le prochain token
L_AR = -sum log P(x_t | x_1, ..., x_{t-1})

MDLM : predire les tokens masques en parallele
L_MDLM = -E_t E_mask [ sum_{i masques} log P(x_i | x_non_masques, t) ]

Block Diffusion (BD3-LM) : divise la sequence en blocs de taille D. Chaque bloc est
demasque en parallele, les blocs sont traites gauche-droite. C’est le meilleur compromis
: generation parallele DANS chaque bloc, structure sequentielle ENTRE les blocs.
Seul changement architectural : masque d’attention causal -> bidirectionnel. Les
poids du modele restent identiques. C’est litteralement une ligne de code.

from transformers import (

import numpy as np

class A2DConfig:

output_dir: str = "./orret-dllm-7b"
block_size: int = 32
total_steps: int = 6000

# Learning rates differencies (cle de SCP)

def patch_attention_to_bidirectional(model):
"""
CHANGEMENT CLE : masque causal -> bidirectionnel.
C'est le seul changement architectural pour la conversion A2D.
"""
patched = 0
for name, module in model.named_modules():

if hasattr(module, 'is_causal'):
module.is_causal = False
patched += 1
if hasattr(model.config, 'is_causal'):
model.config.is_causal = False
print(f"

{patched} modules patches : causal -> bidirectionnel")

return model

class MDLMDataset(Dataset):
"""Dataset MDLM depuis JSONL."""
def __init__(self, data_path, tokenizer, max_seq_len=2048, mask_token_id=None):
self.tokenizer = tokenizer
self.max_seq_len = max_seq_len

self.mask_token_id = mask_token_id or tokenizer.mask_token_id or tokenizer.pad_token_
self.examples = self._load(data_path)
print(f"

Dataset: {len(self.examples)} exemples")

def _load(self, path):
examples = []
with open(path) as f:
for line in f:
if not line.strip(): continue
obj = json.loads(line)
if 'messages' in obj:
text = self.tokenizer.apply_chat_template(
obj['messages'], tokenize=False, add_generation_prompt=False)
elif 'text' in obj:
text = obj['text']
elif 'instruction' in obj:
text = f"{obj['instruction']}\n{obj.get('output','')}"
else:
continue

tokens = self.tokenizer.encode(text, truncation=True, max_length=self.max_seq
if len(tokens) > 16:
examples.append(tokens)
return examples
def __len__(self): return len(self.examples)
def __getitem__(self, idx): return torch.tensor(self.examples[idx], dtype=torch.long)

def collate_mdlm(batch, pad_id, mask_id, mask_min, mask_max, max_len):
"""Collate avec masquage aleatoire pour MDLM."""
L = min(max(len(x) for x in batch), max_len)
input_ids = torch.full((len(batch), L), pad_id, dtype=torch.long)

labels = torch.full((len(batch), L), -100, dtype=torch.long)
attn_mask = torch.zeros(len(batch), L, dtype=torch.long)
for i, tokens in enumerate(batch):
tokens = tokens[:L]
n = len(tokens)
input_ids[i, :n] = tokens
attn_mask[i, :n] = 1
t = np.random.uniform(mask_min, mask_max)
n_mask = max(1, int(n * t))
mask_idx = np.random.choice(n, n_mask, replace=False)
labels[i, mask_idx] = tokens[mask_idx]
input_ids[i, mask_idx] = mask_id
return {'input_ids': input_ids, 'attention_mask': attn_mask, 'labels': labels}

def create_optimizer(model, cfg: A2DConfig):
"""Optimiseur avec LR differencies par type de couche (cle de SCP)."""
attn_params, ffn_params, emb_params, other_params = [], [], [], []
for name, param in model.named_parameters():
if not param.requires_grad: continue
n = name.lower()
if any(x in n for x in ['q_proj','k_proj','v_proj','o_proj','attn','attention']):
attn_params.append(param)
elif any(x in n for x in ['embed','wte','wpe']):
emb_params.append(param)
elif any(x in n for x in ['mlp','ffn','gate','up_proj','down_proj','fc']):
ffn_params.append(param)
else:
other_params.append(param)
for g_name, g_params, lr in [
('attention', attn_params, cfg.lr_attention),
('ffn', ffn_params, cfg.lr_ffn),
('embeddings', emb_params, cfg.lr_embeddings),
('other', other_params, cfg.lr_ffn),
]:
n_p = sum(p.numel() for p in g_params)
print(f"

{g_name:12s}: {n_p/1e6:.1f}M params @ lr={lr:.1e}")

return torch.optim.AdamW([
{'params': attn_params, 'lr': cfg.lr_attention},
{'params': ffn_params,

'lr': cfg.lr_ffn},

{'params': emb_params,

'lr': cfg.lr_embeddings},

{'params': other_params,'lr': cfg.lr_ffn},

], weight_decay=0.01, betas=(0.9, 0.95), eps=1e-8)

def train_a2d(config: A2DConfig, data_path: str):
"""Boucle d'entraînement A2D principale."""
print("=" * 60)
print("SOE-ORRET : CONVERSION AR -> dLLM (A2D)")
print("=" * 60)
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
dtype = torch.bfloat16 if config.__class__.__name__ else torch.float16
if device.type == "cuda":
dtype = torch.bfloat16 if torch.cuda.is_bf16_supported() else torch.float16
else:
dtype = torch.float32
print(f"Device: {device} | Dtype: {dtype}")
# Tokenizer
tokenizer = AutoTokenizer.from_pretrained(config.base_model, trust_remote_code=True)
if tokenizer.mask_token is None:
tokenizer.add_special_tokens({'mask_token': '[MASK]'})
# Modele AR
print(f"Chargement: {config.base_model}")
model = AutoModelForCausalLM.from_pretrained(
config.base_model, torch_dtype=dtype,
device_map="auto" if torch.cuda.is_available() else "cpu",
trust_remote_code=True,
)
model.resize_token_embeddings(len(tokenizer))
# PATCH : causal -> bidirectionnel
model = patch_attention_to_bidirectional(model)
# Dataset
dataset = MDLMDataset(data_path, tokenizer, config.max_seq_len)
collate = partial(collate_mdlm,
pad_id=tokenizer.pad_token_id,
mask_id=tokenizer.mask_token_id,
mask_min=config.mask_prob_min,
mask_max=config.mask_prob_max,
max_len=config.max_seq_len,
)
loader = DataLoader(dataset, batch_size=config.batch_size,
shuffle=True, collate_fn=collate, num_workers=2)
# Optimiseur

optimizer = create_optimizer(model, config)
scheduler = get_cosine_schedule_with_warmup(
optimizer, config.warmup_steps, config.total_steps)
# Entraînement
model.train()
step = 0
total_loss = 0.0
data_iter = iter(loader)
print(f"Debut entraînement ({config.total_steps} steps)...")
while step < config.total_steps:
try:
batch = next(data_iter)
except StopIteration:
data_iter = iter(loader)
batch = next(data_iter)
input_ids = batch['input_ids'].to(device)
attn_mask = batch['attention_mask'].to(device)
labels = batch['labels'].to(device)
outputs = model(input_ids=input_ids, attention_mask=attn_mask)
logits = outputs.logits
# Loss MDLM : cross-entropy sur tokens masques uniquement
loss = F.cross_entropy(
logits.view(-1, logits.size(-1)),
labels.view(-1),
ignore_index=-100,
) / config.grad_accumulation
loss.backward()
total_loss += loss.item() * config.grad_accumulation
if (step + 1) % config.grad_accumulation == 0:
torch.nn.utils.clip_grad_norm_(model.parameters(), 1.0)
optimizer.step()
scheduler.step()
optimizer.zero_grad()
step += 1
if step % config.log_every == 0:
avg = total_loss / config.log_every
lr = optimizer.param_groups[0]['lr']

print(f"Step {step:5d}/{config.total_steps} | Loss: {avg:.4f} | LR: {lr:.2e}")
total_loss = 0.0
if step % config.save_every == 0:
ckpt = Path(config.output_dir) / f"checkpoint-{step}"
ckpt.mkdir(parents=True, exist_ok=True)
model.save_pretrained(str(ckpt))
tokenizer.save_pretrained(str(ckpt))
print(f"Checkpoint: {ckpt}")
# Sauvegarde finale
Path(config.output_dir).mkdir(parents=True, exist_ok=True)
model.save_pretrained(config.output_dir)
tokenizer.save_pretrained(config.output_dir)
with open(Path(config.output_dir) / "conversion_info.json", "w") as f:
json.dump({
"type": "A2D_SCP",
"base_model": config.base_model,
"steps": config.total_steps,
"diffusion_method": "MDLM_BlockDiffusion",
"block_size": config.block_size,
}, f, indent=2)
print("=" * 60)
print("CONVERSION A2D TERMINEE")
print(f"Modele sauvegarde: {config.output_dir}")
print("=" * 60)

if __name__ == "__main__":
parser = argparse.ArgumentParser()
parser.add_argument("--model", default="Qwen/Qwen2.5-7B-Instruct")
parser.add_argument("--data", required=True)
parser.add_argument("--output", default="./orret-dllm-7b")
parser.add_argument("--steps", type=int, default=6000)
args = parser.parse_args()
cfg = A2DConfig()
cfg.base_model = args.model
cfg.output_dir = args.output
cfg.total_steps = args.steps
train_a2d(cfg, args.data)

"""

Sampler Block Diffusion pour SOE-Orret.
Algorithme :
1. Encoder le prompt
2. Initialiser la reponse avec [MASK]*N
3. Pour N etapes : passer au modele bidirectionnel, fixer les tokens confiants
4. Retourner la sequence finale
"""
import torch
import torch.nn.functional as F
from transformers import AutoModelForCausalLM, AutoTokenizer
from typing import Optional, List
import time

class BlockDiffusionSampler:
def __init__(self, model_path: str, device: str = "auto",
n_steps: int = 16, block_size: int = 32,
confidence_threshold: float = 0.9):
self.n_steps = n_steps
self.block_size = block_size
self.confidence_threshold = confidence_threshold
self.tokenizer = AutoTokenizer.from_pretrained(model_path, trust_remote_code=True)
self.mask_token_id = self.tokenizer.mask_token_id or self.tokenizer.pad_token_id
if device == "auto":
device = "cuda" if torch.cuda.is_available() else "cpu"
self.device = device
self.model = AutoModelForCausalLM.from_pretrained(
model_path,
torch_dtype=torch.bfloat16 if device == "cuda" else torch.float32,
device_map=device, trust_remote_code=True,
)
self.model.eval()

print(f"[Orret dLLM] Modele charge sur {device} | steps={n_steps} | block={block_size
@torch.no_grad()
def generate(self, prompt: str, max_new_tokens: int = 512,
temperature: float = 0.7, top_p: float = 0.9,
verbose: bool = False) -> str:
t0 = time.time()

n_blocks = (max_new_tokens + self.block_size - 1) // self.block_size
generated_ids = []
for _ in range(n_blocks):

existing = torch.tensor(generated_ids, dtype=torch.long).unsqueeze(0).to(self.dev
if generated_ids else prompt_ids[:, :0]
prefix = torch.cat([prompt_ids, existing], dim=1)
block = self._generate_block(prefix, temperature, top_p, verbose)
generated_ids.extend(block)
if self.tokenizer.eos_token_id in block:
eos = block.index(self.tokenizer.eos_token_id)
generated_ids = generated_ids[:len(generated_ids)-len(block)+eos]
break
if verbose:
t = time.time() - t0

print(f"\n[{len(generated_ids)} tokens en {t:.1f}s = {len(generated_ids)/t:.1f} t
return self.tokenizer.decode(generated_ids, skip_special_tokens=True)
@torch.no_grad()
def _generate_block(self, prompt_ids, temperature, top_p, verbose) -> List[int]:
"""Generation d'un bloc de tokens via demasquage iteratif."""
mask_block = torch.full((1, self.block_size), self.mask_token_id,
dtype=torch.long, device=self.device)
full_seq = torch.cat([prompt_ids, mask_block], dim=1)
mask_positions = list(range(prompt_ids.shape[1], full_seq.shape[1]))
still_masked = set(mask_positions)
fixed_tokens = {}
for step in range(self.n_steps):
if not still_masked: break
outputs = self.model(full_seq)
logits = outputs.logits
for pos in list(still_masked):
probs = F.softmax(logits[0, pos, :] / temperature, dim=-1)
max_prob, best_token = probs.max(dim=-1)
if max_prob.item() >= self.confidence_threshold or step == self.n_steps - 1:
if temperature > 0:
sorted_probs, sorted_idx = probs.sort(descending=True)
cum = sorted_probs.cumsum(dim=-1)
sorted_probs[cum > top_p] = 0
sorted_probs /= sorted_probs.sum() + 1e-8
token_id = sorted_idx[torch.multinomial(sorted_probs, 1)].item()

else:
token_id = best_token.item()
fixed_tokens[pos] = token_id
full_seq[0, pos] = token_id
still_masked.discard(pos)
if verbose:
partial = self.tokenizer.decode(
[fixed_tokens.get(p, self.mask_token_id) for p in mask_positions],
skip_special_tokens=True)
print(f"\r

Step {step+1}/{self.n_steps} | {partial[:60]}", end="", flush=Tru

return [fixed_tokens.get(p, self.mask_token_id) for p in mask_positions]

# ~/soe/memory/hierarchy.py
"""
Memoire hierarchique SOE-Orret — 5 couches inspirées de l'architecture ARIA.
L1 Working Memory

: contexte actif (~4K tokens, RAM pure)

L2 Episodic Memory : experiences passees avec scoring de coherence
L3 Semantic Memory : connaissances factuelles (FAISS)
L4 Procedural

: sequences d'actions et patterns

L5 World Model

: modele interne du monde / self-awareness

"""
import json, time, hashlib
from pathlib import Path
from dataclasses import dataclass, field, asdict
from typing import List, Optional, Dict, Any

@dataclass
class Episode:
id: str
timestamp: float
user_input: str
ai_response: str

outcome: str = "unknown"
coherence_issues: List[str] = field(default_factory=list)
lessons_learned: List[str] = field(default_factory=list)
importance: float = 0.5

class WorkingMemory:
"""Contexte actif de la conversation en cours."""
def __init__(self, max_tokens: int = 4096):
self.max_tokens = max_tokens
self._items = []
self._tokens = 0
def add(self, role: str, content: str):
est_tokens = len(content.split()) * 1.3
self._items.append({"role": role, "content": content})
self._tokens += est_tokens
while self._tokens > self.max_tokens and len(self._items) > 1:
removed = self._items.pop(0)
self._tokens -= len(removed["content"].split()) * 1.3
def get_context(self):
return [{"role": i["role"], "content": i["content"]} for i in self._items]
def clear(self):
self._items.clear()
self._tokens = 0

class EpisodicMemory:
"""
Memoire episodique — experiences avec analyse de coherence.
Implemente la vision ARIA : "l'IA juge ses propres experiences,
detecte les incoherences, et remodele son comportement."
"""
def __init__(self, path: str):
self.path = Path(path)
self.path.mkdir(parents=True, exist_ok=True)
self.file = self.path / "episodes.jsonl"
self.episodes: List[Episode] = self._load()
def _load(self):
eps = []
if self.file.exists():

with open(self.file) as f:
for line in f:
if line.strip():
eps.append(Episode(**json.loads(line)))
return eps
def store(self, episode: Episode):
self.episodes.append(episode)
with open(self.file, "a") as f:
f.write(json.dumps(asdict(episode), ensure_ascii=False) + "\n")
def recall(self, query: str, n: int = 5) -> List[Episode]:
"""Recupere les episodes les plus recents et importants."""
return sorted(self.episodes,
key=lambda e: e.timestamp * 0.3 + e.importance * 0.7,
reverse=True)[:n]
def analyze_coherence(self) -> Dict:
"""
Analyse les patterns d'erreur recurrents.
C'est la vision 'retourner dans le passe' de tes notes ARIA.
"""
if not self.episodes:
return {"coherence_avg": 1.0, "patterns": [], "lessons": []}
all_issues = []
for ep in self.episodes:
all_issues.extend(ep.coherence_issues)
freq = {}
for issue in all_issues:
freq[issue] = freq.get(issue, 0) + 1
recurring = [i for i, c in freq.items() if c >= 2]
all_lessons = []
for ep in self.episodes:
all_lessons.extend(ep.lessons_learned)
return {
"n_episodes": len(self.episodes),
"recurring_patterns": recurring,
"lessons": list(dict.fromkeys(all_lessons))[:10],
}

class OrretMemorySystem:
"""Gestionnaire unifie de la memoire hierarchique."""

def __init__(self, base_path: str = "~/soe/memory"):
self.base = Path(base_path).expanduser()
self.working = WorkingMemory(max_tokens=4096)
self.episodic = EpisodicMemory(str(self.base / "episodic"))
print(f"Memoire ARIA: {len(self.episodic.episodes)} episodes")
def retrieve_context(self, query: str) -> str:
"""Contexte memoriel pour enrichir le prompt."""
parts = []
eps = self.episodic.recall(query, n=3)
if eps:
parts.append("## Experiences passees pertinentes")
for ep in eps:
lesson = ep.lessons_learned[0] if ep.lessons_learned else ""
parts.append(f"- [{ep.outcome}] '{ep.user_input[:60]}' -> {lesson}")
analysis = self.episodic.analyze_coherence()
if analysis["recurring_patterns"]:
parts.append("## Points de vigilance")
for p in analysis["recurring_patterns"][:2]:
parts.append(f"- {p}")
return "\n".join(parts)
def record(self, user_input: str, response: str, outcome: str = "success"):
ep = Episode(
id=hashlib.sha256(f"{user_input}{time.time()}".encode()).hexdigest()[:16],
timestamp=time.time(),
user_input=user_input,
ai_response=response,
outcome=outcome,
importance=min(0.5 + len(response)/2000, 1.0),
)
self.episodic.store(ep)
self.working.add("assistant", response)

def web_search(query: str, n: int = 2) -> str:
"""Recherche web simple avec DuckDuckGo."""
if not HAS_DDG:
return ""
results = []
try:
with DDGS() as ddgs:
for r in ddgs.text(query, max_results=n):

def needs_web(user_input: str) -> bool:
"""Detecte si une recherche web est necessaire."""
keywords = ['aujourd hui','recent','derniere','actualite','2025','2026',
'version','maintenant','actuel','nouveau','vient de',
'prix','disponible','qui est','quel est','combien']
return any(kw in user_input.lower() for kw in keywords)

class OrretAgent:
def __init__(self, sampler, memory_system,
enable_web=True, enable_reflection=True, verbose=False):
self.sampler = sampler
self.memory = memory_system
self.enable_web = enable_web
self.enable_reflection = enable_reflection
self.verbose = verbose
def _build_prompt(self, user_input: str, web_ctx: str = "") -> str:
memory_ctx = self.memory.retrieve_context(user_input)
system = SYSTEM_PROMPT.format(memory_context=memory_ctx).strip()
messages = [{"role": "system", "content": system}]
if web_ctx:
messages.append({"role": "system", "content": f"[CONTEXTE WEB]\n{web_ctx}"})
messages.extend(self.memory.working.get_context())
messages.append({"role": "user", "content": user_input})
return self.sampler.tokenizer.apply_chat_template(
messages, tokenize=False, add_generation_prompt=True)
def _critique(self, question: str, response: str) -> bool:
"""Autocritique rapide : la reponse est-elle coherente ?"""
prompt = f"""Critique en 3 mots max. "OK" si la reponse est correcte.
Question: {question[:100]}
Reponse: {response[:200]}
Critique:"""
critique = self.sampler.generate(prompt, max_new_tokens=30, temperature=0.3)
return "OK" not in critique.upper() and len(critique.strip()) > 2
def run(self, user_input: str) -> str:
"""Pipeline : RAG optionnel -> dLLM -> autocritique -> memoire."""
self.memory.working.add("user", user_input)

# RAG web si necesaire
web_ctx = ""
if self.enable_web and needs_web(user_input):
if self.verbose: print(f"[RAG] Recherche: {user_input[:50]}")
web_ctx = web_search(user_input)
# Generation
prompt = self._build_prompt(user_input, web_ctx)
response = self.sampler.generate(prompt, max_new_tokens=512, temperature=0.7)
# Gerer auto-demande de recherche
if "[RECHERCHE:" in response and self.enable_web:
m = re.search(r'\[RECHERCHE:\s*(.*?)\]', response)
if m:
extra_ctx = web_search(m.group(1).strip())
prompt2 = self._build_prompt(user_input, web_ctx + "\n" + extra_ctx)
response = self.sampler.generate(prompt2, max_new_tokens=512)
# Autocritique simple
if self.enable_reflection and self._critique(user_input, response):
if self.verbose: print("[Critique] Revision...")
prompt_rev = self._build_prompt(
user_input + "\n[Note interne: ameliore la precision]", web_ctx)

response = self.sampler.generate(prompt_rev, max_new_tokens=512, temperature=0.5)
self.memory.record(user_input, response, "success")
return response
